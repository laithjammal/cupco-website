# Ad landing page for new.cupco.com.au.
#
# Crawling is ALLOWED on purpose. The page carries a <meta name="robots"
# content="noindex"> tag to keep it out of organic search results so it does
# not compete with cupco.com.au. Google can only obey that tag if it is
# allowed to fetch the page and read it — blocking crawlers here would hide
# the noindex and let the bare URL get indexed anyway.
#
# AdsBot-Google is named explicitly so Google Ads can always reach the page
# for landing-page quality checks.

User-agent: *
Allow: /

User-agent: AdsBot-Google
Allow: /

User-agent: AdsBot-Google-Mobile
Allow: /
